<script>
    // ============================================================
    // 1. 折线图配置（表1）
    //    请将 xData / yData 替换为表1的真实数据
    // ============================================================
    var lineChart = echarts.init(document.getElementById('lineChart'));

    // ---- 替换为表1的实际数据 ----
    var xData = ['类别A', '类别B', '类别C', '类别D', '类别E', '类别F'];
    var yData = [120, 200, 150, 80, 170, 230];
    // 如果需要“比例”，请在此处计算每个点相对于总数的百分比
    var total = yData.reduce(function (a, b) { return a + b; }, 0);

    var lineOption = {
        color: ['#1a56db'],                // 主蓝色
        tooltip: {
            trigger: 'axis',
            backgroundColor: '#ffffff',
            borderColor: '#1a56db',
            borderWidth: 1,
            textStyle: { color: '#000000', fontSize: 13 },
            // 自定义提示框：同时显示“数值”和“比例”
            formatter: function (params) {
                var p = params[0];
                var value = p.value;
                var percent = total > 0 ? (value / total * 100).toFixed(1) : '0.0';
                return (
                    '<b>' + p.name + '</b><br/>' +
                    '数值：' + value + '<br/>' +
                    '比例：' + percent + '%'
                );
            }
        },
        grid: { left: '8%', right: '6%', top: '12%', bottom: '12%', containLabel: true },
        xAxis: {
            type: 'category',
            data: xData,
            axisLine: { lineStyle: { color: '#9ca3af' } },
            axisLabel: { color: '#000000', fontSize: 13 },
            axisTick: { show: false }
        },
        yAxis: {
            type: 'value',
            name: '数值',
            nameTextStyle: { color: '#374151', fontSize: 13 },
            axisLine: { show: false },
            axisLabel: { color: '#000000', fontSize: 13 },
            splitLine: { lineStyle: { color: '#e5e7eb', type: 'dashed' } }
        },
        series: [{
            name: '表1数据',
            type: 'line',
            smooth: true,                  // 平滑曲线，动态感更强
            symbol: 'circle',
            symbolSize: 10,                // 数据点清晰可见
            showAllSymbol: true,           // 强制显示所有数据点
            lineStyle: { width: 3, color: '#1a56db' },
            itemStyle: {
                color: '#1a56db',
                borderColor: '#ffffff',
                borderWidth: 2
            },
            areaStyle: {                   // 浅蓝色面积填充，突出主色调
                color: {
                    type: 'linear',
                    x: 0, y: 0, x2: 0, y2: 1,
                    colorStops: [
                        { offset: 0, color: 'rgba(26, 86, 219, 0.25)' },
                        { offset: 1, color: 'rgba(26, 86, 219, 0.02)' }
                    ]
                }
            },
            data: yData,
            // 在数据点上方显示数值（可选，如不需要可删去 label 整段）
            label: {
                show: false,               // 设为 true 可直接在点上显示数字
                position: 'top',
                color: '#1a56db',
                fontWeight: 'bold',
                fontSize: 12
            }
        }]
    };

    lineChart.setOption(lineOption);

    // ============================================================
    // 2. 柱形图配置（图1）
    //    请将 barXData / barYData 替换为图1的真实数据
    // ============================================================
    var barChart = echarts.init(document.getElementById('barChart'));

    // ---- 替换为图1的实际数据 ----
    var barXData = ['项目一', '项目二', '项目三', '项目四', '项目五'];
    var barYData = [320, 480, 260, 590, 410];

    var barOption = {
        color: ['#1a56db'],
        tooltip: {
            trigger: 'axis',
            axisPointer: { type: 'shadow' },
            backgroundColor: '#ffffff',
            borderColor: '#1a56db',
            borderWidth: 1,
            textStyle: { color: '#000000', fontSize: 13 },
            formatter: function (params) {
                var p = params[0];
                return '<b>' + p.name + '</b><br/>数值：' + p.value;
            }
        },
        grid: { left: '8%', right: '6%', top: '12%', bottom: '12%', containLabel: true },
        xAxis: {
            type: 'category',
            data: barXData,
            axisLine: { lineStyle: { color: '#9ca3af' } },
            axisLabel: { color: '#000000', fontSize: 13, interval: 0 },
            axisTick: { show: false }
        },
        yAxis: {
            type: 'value',
            name: '数值',
            nameTextStyle: { color: '#374151', fontSize: 13 },
            axisLine: { show: false },
            axisLabel: { color: '#000000', fontSize: 13 },
            splitLine: { lineStyle: { color: '#e5e7eb', type: 'dashed' } }
        },
        series: [{
            name: '图1数据',
            type: 'bar',
            barWidth: '50%',
            itemStyle: {
                color: {
                    type: 'linear',
                    x: 0, y: 0, x2: 0, y2: 1,
                    colorStops: [
                        { offset: 0, color: '#3b82f6' },
                        { offset: 1, color: '#1a56db' }
                    ]
                },
                borderRadius: [4, 4, 0, 0]   // 顶部圆角，视觉更柔和
            },
            data: barYData,
            // 柱顶显示数值（如不需要可设为 false）
            label: {
                show: true,
                position: 'top',
                color: '#1a56db',
                fontWeight: 'bold',
                fontSize: 12
            }
        }]
    };

    barChart.setOption(barOption);

    // ============================================================
    // 3. 响应式：窗口大小变化时重绘
    // ============================================================
    window.addEventListener('resize', function () {
        lineChart.resize();
        barChart.resize();
    });
</script>
